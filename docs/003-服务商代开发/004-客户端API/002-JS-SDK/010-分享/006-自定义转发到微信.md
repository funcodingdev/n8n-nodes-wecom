## 旧版jweixin调用
### 示例代码
```javascript
wx.invoke(
 "shareWechatMessage", {
 title: '', // 分享标题
 desc: '', // 分享描述
 link: '', // 分享链接
 imgUrl: '', // 分享封面
 enableIdTrans: 1, // 是否开启id转译，不填默认为0
 }, function(res) {
 if (res.err_msg == "shareWechatMessage:ok") {
 //正确处理
 }else {
 //错误处理
 }
 }
);
```
